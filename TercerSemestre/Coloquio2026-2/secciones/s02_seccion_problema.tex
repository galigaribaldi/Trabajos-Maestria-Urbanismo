      {\small\bfseries\color{ColorPrincipal}
        El Periférico conecta en el plano,\\pero no en la red.}\\[0.2cm]
      {\scriptsize
        Fue diseñado como anillo urbano, pero opera hoy como autopista
        exclusiva del automóvil privado. Ninguna línea de transporte
        masivo lo recorre.\\[0.2cm]
        La paradoja no es solo del sur: \textbf{cualquier viaje entre
        zonas periféricas obliga a pasar por el centro}, sin importar
        el contorno de origen.\\[0.2cm]
        Xochimilco, Tláhuac y Milpa Alta son el caso extremo —
        pero Ecatepec, Neza y Naucalpan padecen la misma lógica radial.
      }\\[0.2cm]
      {\color{ColorAcento}\rule{0.55\textwidth}{0.8pt}}
    \end{column}
  \end{columns}

\begin{frame}{La Red Óptima Necesita un Anillo}
  \begin{columns}[c]
    % ── Columna izquierda: red radial ────────────────────────────────────────
    \begin{column}{0.27\textwidth}
      \centering
      \begin{tikzpicture}[
        nodo/.style   = {circle, draw=ColorPrincipal, fill=ColorPrincipal!15,
                         minimum size=0.45cm, line width=0.7pt},
        centro/.style = {circle, draw=ColorPrincipal, fill=ColorPrincipal,
                         minimum size=0.55cm, line width=0.7pt},
        arista/.style = {draw=ColorPrincipal!70, line width=0.9pt},
      ]
        \node[centro] (c) at (0,0) {};
        \foreach \i in {0,1,2,3,4,5}{
          \node[nodo] (p\i) at ({60*\i}:1.3cm) {};
          \draw[arista] (c) -- (p\i);
        }
      \end{tikzpicture}\\[0.2cm]
      {\small\bfseries\color{ColorPrincipal} Red Radial}\\[0.08cm]
      {\scriptsize Óptima para redes\\
        pequeñas. Todo viaje\\
        pasa por el centro.}
    \end{column}
    % ── Columna central: teorema citado ──────────────────────────────────────
    \begin{column}{0.42\textwidth}
      \begin{exampleblock}{La Topología Óptima}
        \centering
        {\scriptsize Toda red de transporte supera un umbral\\
          crítico de longitud total:}\\[0.12cm]
        {\LARGE\bfseries\color{ColorAcento} $L_c$}\\[0.12cm]
        {\scriptsize
          A partir de ese punto, la topología que\\
          \textbf{minimiza el tiempo promedio de viaje}\\
          incorpora obligatoriamente un anillo.\\[0.1cm]
          Una red sin anillo es, por definición,\\
          \textbf{subóptima}.}\\[0.12cm]
        {\scriptsize\itshape\citet{aldous2019optimal}}
      \end{exampleblock}
    \end{column}
    % ── Columna derecha: red con anillo ──────────────────────────────────────
    \begin{column}{0.27\textwidth}
      \centering
      \begin{tikzpicture}[
        nodo/.style   = {circle, draw=ColorPrincipal, fill=ColorPrincipal!15,
                         minimum size=0.45cm, line width=0.7pt},
        centro/.style = {circle, draw=ColorPrincipal, fill=ColorPrincipal,
                         minimum size=0.55cm, line width=0.7pt},
        arista/.style = {draw=ColorPrincipal!70, line width=0.9pt},
        anillo/.style = {draw=ColorAcento, line width=1.8pt},
      ]
        \node[centro] (c) at (0,0) {};
        \foreach \i in {0,1,2,3,4,5}{
          \node[nodo] (p\i) at ({60*\i}:1.3cm) {};
          \draw[arista] (c) -- (p\i);
        }
        \draw[anillo] (p0)--(p1)--(p2)--(p3)--(p4)--(p5)--cycle;
      \end{tikzpicture}\\[0.2cm]
      {\small\bfseries\color{ColorPrincipal} Red con Anillo}\\[0.08cm]
      {\scriptsize Óptima cuando la red\\
        supera~$L_c$. Viajes\\
        periférico-periférico directos.}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  {\small\color{ColorPrincipal}
    La \zmvm\ ya superó ese umbral —
    el Periférico sin transporte masivo es, matemáticamente, una red subóptima.}
\end{frame}
