% ── Frame 4.5: Conceptos Clave del Modelo ────────────────────────────────────
\begin{frame}{Conceptos Clave: El Lenguaje del Modelo}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\scriptsize\bfseries Eficiencia de Red}
        {\scriptsize
          Capacidad de la red para conectar todos sus nodos
          minimizando el \textbf{tiempo de viaje promedio}.\\[0.08cm]
          Una red eficiente lleva a más personas en menos tiempo.}
      \end{block}
      \vspace{0.15cm}
      \begin{block}{\scriptsize\bfseries Vulnerabilidad}
        {\scriptsize
          Cuánto \textbf{cae la eficiencia global} al eliminar
          un nodo o una arista crítica.\\[0.08cm]
          Una red vulnerable depende de pocos puntos para no colapsar.}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\scriptsize\bfseries Interconectividad}
        {\scriptsize
          Grado en que los nodos pueden alcanzarse
          \textbf{mutuamente sin depender de un único punto}.\\[0.08cm]
          Baja interconectividad = todos los caminos pasan
          por el mismo nodo central.}
      \end{block}
      \vspace{0.15cm}
      \begin{block}{\scriptsize\bfseries Fuerza Capilar}
        {\scriptsize
          Concentración de flujos en un nodo periférico
          \textbf{sin alternativa lateral}.\\[0.08cm]
          Alta Fuerza Capilar indica un cuello de botella
          invisible en la periferia.}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.1cm}
  {\scriptsize\itshape\color{ColorPrincipal}
    Estos conceptos son el lenguaje del VFTModel —
    los indicadores los operacionalizan con datos reales de la \zmvm{}.}
\end{frame}
