% ── \InfoCabecera ─────────────────────────────────────────────────────────────
% Encabezado institucional (franja ColorPrincipal + escudos + título)
% Se coloca al inicio del entorno tikzpicture principal.
%
% Uso: \InfoCabecera (sin argumentos — lee \TituloInfografia, \Materia, etc.)
%
\newcommand{\InfoCabecera}{%

  \fill[ColorPrincipal]
    (0, \dimexpr\paperheight-1.6cm\relax)
    rectangle (\dimexpr\paperwidth-1.6cm\relax, \dimexpr\paperheight-1.6cm-2.4cm\relax);
  % Línea de acento inferior
  \fill[ColorAcento]
    (0, \dimexpr\paperheight-1.6cm-2.4cm\relax)
    rectangle (\dimexpr\paperwidth-1.6cm\relax, \dimexpr\paperheight-1.6cm-2.4cm-3pt\relax);
  % Escudo UNAM izquierda
  \node[anchor=west, inner sep=0pt] at
    (0.3cm, \dimexpr\paperheight-1.6cm-1.2cm\relax)
    {\includegraphics[height=1.6cm]{UNAM_crest_blue.pdf}};
  % Escudo FES derecha
  \node[anchor=east, inner sep=0pt] at
    (\dimexpr\paperwidth-1.6cm-0.3cm\relax, \dimexpr\paperheight-1.6cm-1.2cm\relax)
    {\includegraphics[height=1.6cm]{escudo-fes-azul.png}};
  % Texto central
  \node[anchor=center, align=center, text=white] at
    ({\dimexpr(\paperwidth-1.6cm)/2\relax}, \dimexpr\paperheight-1.6cm-0.55cm\relax)
    {{\tiny\textbf{UNIVERSIDAD NACIONAL AUTÓNOMA DE MÉXICO}} \\[1pt]
     {\tiny\textbf{MAESTRÍA EN URBANISMO — FES ACATLÁN}}};
  \node[anchor=center, align=center] at
    ({\dimexpr(\paperwidth-1.6cm)/2\relax}, \dimexpr\paperheight-1.6cm-1.55cm\relax)

% ── \InfoPie ──────────────────────────────────────────────────────────────────
% Pie de página institucional (franja delgada)
%
% Uso: \InfoPie{Texto de fuente}
%
\newcommand{\InfoPie}[1]{%
  \fill[ColorPrincipal!85!black]
    (0, 0)
    rectangle (\dimexpr\paperwidth-1.6cm\relax, 0.7cm);
  \fill[ColorAcento]
    (0, 0.7cm)
    rectangle (\dimexpr\paperwidth-1.6cm\relax, 0.7cm+2pt);
  \node[anchor=west, text=white, font=\tiny] at
    (0.3cm, 0.35cm)
    {#1};
  \node[anchor=east, text=white!70!ColorPrincipal, font=\tiny] at
    (\dimexpr\paperwidth-1.6cm-0.3cm\relax, 0.35cm)
    {\Materia\ —\ \FechaInfografia};
}

% ── \InfoImagenPlaceholder ────────────────────────────────────────────────────
% Rectángulo placeholder para imagen + pie de fuente
%
% Argumentos:
%   #1 = ancho (ej. \linewidth, 5cm, 0.9\linewidth)
%   #2 = alto  (ej. 4cm, 5cm)
%   #3 = texto del pie de fuente
%
% Uso: \InfoImagenPlaceholder{\linewidth}{4cm}{Fuente: Elaboración propia}
%
\newcommand{\InfoImagenPlaceholder}[3]{%
  \begin{tikzpicture}
    \node[img placeholder={#1}, minimum width=#1, minimum height=#2]
      {\faImage\\[4pt]\textit{[Insertar imagen]}};
  \end{tikzpicture}%
  \par{\tiny\itshape Fuente: #3}\par%
}


% ── \InfoDato ─────────────────────────────────────────────────────────────────
% Bloque de dato clave / KPI: número grande + etiqueta + descripción
%
% Argumentos:
%   #1 = número o valor grande (ej. 42\%, 1.2M, 2050)
%   #2 = etiqueta corta       (ej. "de cobertura arbórea")
%   #3 = descripción breve    (ej. "En zonas de alta densidad")
%   #4 = ícono fontawesome    (ej. \faTree, \faCity, \faLeaf)
%
% Uso: \InfoDato{42\%}{cobertura arbórea}{Zonas de alta densidad}{\faTree}
%
\newcommand{\InfoDato}[4]{%
  \begin{tcolorbox}[
    enhanced,
    colback  = ColorFondoClaro,
    colframe = ColorPrincipal,
    boxrule  = 0.6pt,
    arc      = 4pt,
    top=4pt, bottom=4pt, left=6pt, right=6pt,
    shadow   = {1pt}{-1pt}{0pt}{black!10}
  ]
    \begin{center}
      {\color{ColorPrincipal}\large #4}\\[2pt]
      {\fontsize{28}{30}\selectfont\bfseries\color{ColorAcento} #1}\\[1pt]
      {\small\bfseries\color{ColorPrincipal} #2}\\[2pt]
      {\footnotesize\color{ColorTexto} #3}
    \end{center}
  \end{tcolorbox}%
}


% ── \InfoIconoItem ────────────────────────────────────────────────────────────
% Item de lista con ícono fontawesome a la izquierda
%
% Uso: \InfoIconoItem{\faLeaf}{Texto del ítem}
%
\newcommand{\InfoIconoItem}[2]{%
  \noindent
  \begin{minipage}[t]{0.06\linewidth}
    \centering\color{ColorAcento}\normalsize #1
  \end{minipage}%
  \begin{minipage}[t]{0.92\linewidth}
    \footnotesize\color{ColorTexto} #2
  \end{minipage}\\[4pt]%
}
