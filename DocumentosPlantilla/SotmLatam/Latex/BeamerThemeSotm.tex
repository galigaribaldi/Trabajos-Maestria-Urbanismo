% =============================================================================
%  4. FOOTLINE — Autor | SOTM Latam CDMX 2026 | N/Total
% =============================================================================
\makeatletter
\setbeamertemplate{footline}{%
  \leavevmode%
  \hbox{%
    \begin{beamercolorbox}[wd=.45\paperwidth, ht=2.5ex, dp=1.5ex, leftskip=0.4cm]{palette primary}%
      \usebeamerfont{author in head/foot}\AutorCorto
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=.40\paperwidth, ht=2.5ex, dp=1.5ex, center]{palette secondary}%
      \usebeamerfont{title in head/foot}SOTM Latam CDMX 2026
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=.15\paperwidth, ht=2.5ex, dp=1.5ex, rightskip=0.4cm, right]{palette primary}%
      \usebeamerfont{date in head/foot}%
      \insertframenumber{} / \inserttotalframenumber
    \end{beamercolorbox}%
  }%
  \vskip0pt%
}
\makeatother


% =============================================================================
%  5. HEADLINE — eliminado
% =============================================================================
\setbeamertemplate{headline}{}


% =============================================================================
%  6. FRAMETITLE — variante simple y variante con línea terracota
% =============================================================================
\usepackage{etoolbox}

\defbeamertemplate{frametitle}{simple}{%
  \begin{beamercolorbox}[wd=\paperwidth, sep=0.20cm, left, leftskip=0.20cm]{frametitle}%
    {\tiny\color{white!60!SotmVerde}\insertsectionhead}\\[-0.05cm]%
    \bfseries\insertframetitle\strut%
  \end{beamercolorbox}%
}

\defbeamertemplate{frametitle}{conlinea}{%
  \begin{beamercolorbox}[wd=\paperwidth, sep=0.20cm, dp=0cm, left, leftskip=0.20cm]{frametitle}%
    {\tiny\color{white!60!SotmVerde}\insertsectionhead}\\[-0.05cm]%
    \bfseries\insertframetitle\strut%
  \end{beamercolorbox}%
  \vspace{-0.06cm}%
  {\color{SotmTerracota}\rule{\paperwidth}{2.5pt}}%
}

\setbeamertemplate{frametitle}[simple]

\newif\ifusarlineaterracota
\usarlineaterracotafalse

\AtBeginEnvironment{frame}{%
  \ifusarlineaterracota
    \setbeamertemplate{frametitle}[conlinea]%
    \global\usarlineaterracotafalse
  \else
    \setbeamertemplate{frametitle}[simple]%
  \fi
}

% \framelinea — activa la línea terracota para el siguiente frame
\newcommand{\framelinea}{\global\usarlineaterracotatrue}% activa línea terracota

        \vfill
        {\color{SotmTerracota}\rule{0.6\textwidth}{1.5pt}}\\[0.4cm]
        {\Large\bfseries\color{SotmVerde}\insertsectionhead}\\[0.4cm]
        {\color{SotmTerracota}\rule{0.6\textwidth}{1.5pt}}
        \vfill
      \end{center}
    \end{frame}
  \fi
}
