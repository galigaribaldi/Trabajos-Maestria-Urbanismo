\begin{frame}{Metodología — Diagrama de Flujo}
  \begin{center}
  \begin{tikzpicture}[
    caja/.style     = {rectangle, rounded corners=5pt,
                       draw=SotmVerde, line width=0.8pt,
                       fill=SotmCrema, text width=2.6cm,
                       align=center, minimum height=0.75cm, font=\small},
    inicio/.style   = {caja, fill=SotmVerde, text=white, font=\small\bfseries},
    result/.style   = {caja, fill=SotmTerracota, text=white, font=\small\bfseries},
    decision/.style = {diamond, draw=SotmVerde, line width=0.8pt,
                       fill=SotmCrema, aspect=2.5,
                       text width=2.2cm, align=center, font=\scriptsize},
    flecha/.style   = {-Stealth, thick, color=SotmVerde},
    lateral/.style  = {-Stealth, thick, color=SotmTerracota, dashed},
    node distance   = 0.55cm and 2.2cm
  ]
    \node[inicio]              (ini)  {Datos de entrada};
    \node[caja, below=of ini]  (proc) {Procesamiento};
    \node[decision, below=of proc](dec) {¿Válido?};
    \node[caja, below=of dec]  (anal) {Análisis};
    \node[result, below=of anal](res) {Resultado};

    \draw[flecha] (ini)  -- (proc);
    \draw[flecha] (proc) -- (dec);
    \draw[flecha] (dec)  -- node[right, font=\scriptsize]{Sí} (anal);
    \draw[flecha] (anal) -- (res);

    \node[caja, right=of dec] (corr) {Corrección};
    \draw[lateral] (dec)  -- node[above, font=\scriptsize]{No} (corr);
    \draw[lateral] (corr) |- (ini);
  \end{tikzpicture}
  \end{center}
\end{frame}
