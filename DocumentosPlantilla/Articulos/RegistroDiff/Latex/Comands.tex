% Figura portátil: muestra la ruta del archivo como evidencia documental sin
% necesidad de distribuir la imagen con el repositorio.
% Uso: \figplaceholder{Figures/Mapas/Mapa_Red_Esquematica.pdf}
\newcommand{\figplaceholder}[1]{%
  \begin{tikzpicture}
    \node[draw=ColorPrincipal!40, dashed, fill=gray!4, rounded corners=5pt,
          text width=\dimexpr\textwidth-2.4cm\relax, minimum height=4cm,
          align=center, inner sep=14pt]
      {\small\color{gray!55}\textbf{[Figura referenciada --- archivo externo]}\\[8pt]
       \footnotesize\color{ColorPrincipal!70}\nolinkurl{#1}};
  \end{tikzpicture}%
}

% Figura condicional: incluye el archivo si existe; si no, muestra \figplaceholder.
% Reemplaza \includegraphics en diffs generados por latexdiff-tesis.
% Uso: \figinclude[width=\textwidth]{Figures/Mapas/Mapa_Red_Esquematica}
\newcommand{\figinclude}[2][width=\textwidth]{%
  \IfFileExists{#2.pdf}{\includegraphics[#1]{#2}}{%
  \IfFileExists{#2.png}{\includegraphics[#1]{#2}}{%
  \IfFileExists{#2}    {\includegraphics[#1]{#2}}{%
    \figplaceholder{#2}%
  }}}%
}
