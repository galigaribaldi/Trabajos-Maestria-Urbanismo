% =============================================================================
%  latexdiff-preamble.tex — Definiciones de marcado para latexdiff
%
%  Este archivo es cargado por main.tex DESPUÉS de \input{Latex/Comands}
%  y ANTES de \usepackage{hyperref}.
%
%  ESTILO DE MARCADO: Color puro (sin decoraciones ulem)
%  ──────────────────────────────────────────────────────
%  Se usa \color{blue} para adiciones y \color{red} para eliminaciones,
%  en lugar de \uwave (subrayado ondulado) y \sout (tachado).
%
%  Motivo: ulem procesa el texto carácter a carácter y falla cuando
%  \DIFadd{} o \DIFdel{} contienen comandos con grupos propios
%  (p.ej. \textbf{}, macros con argumentos). El color puro es
%  compatible con cualquier contenido LaTeX arbitrario.
% =============================================================================

%DIF UNDERLINE PREAMBLE %DIF PREAMBLE
\RequirePackage[normalem]{ulem} %DIF PREAMBLE  (cargado por compatibilidad)
\RequirePackage{color}\definecolor{RED}{rgb}{1,0,0}\definecolor{BLUE}{rgb}{0,0,1} %DIF PREAMBLE
\providecommand{\DIFadd}[1]{{\protect\color{blue}#1}}   %DIF PREAMBLE
\providecommand{\DIFdel}[1]{{\protect\color{red}#1}}    %DIF PREAMBLE
%DIF SAFE PREAMBLE %DIF PREAMBLE
\providecommand{\DIFaddbegin}{} %DIF PREAMBLE
\providecommand{\DIFaddend}{} %DIF PREAMBLE
\providecommand{\DIFdelbegin}{} %DIF PREAMBLE
\providecommand{\DIFdelend}{} %DIF PREAMBLE
\providecommand{\DIFmodbegin}{} %DIF PREAMBLE
\providecommand{\DIFmodend}{} %DIF PREAMBLE
%DIF FLOATSAFE PREAMBLE %DIF PREAMBLE
\providecommand{\DIFaddFL}[1]{\DIFadd{#1}} %DIF PREAMBLE
\providecommand{\DIFdelFL}[1]{\DIFdel{#1}} %DIF PREAMBLE
\providecommand{\DIFaddbeginFL}{} %DIF PREAMBLE
\providecommand{\DIFaddendFL}{} %DIF PREAMBLE
\providecommand{\DIFdelbeginFL}{} %DIF PREAMBLE
\providecommand{\DIFdelendFL}{} %DIF PREAMBLE
% Compatibilidad biblatex → natbib: el diff puede contener \parencite/\textcite
\providecommand{\parencite}[2][]{\citep{#2}}
\providecommand{\textcite}[2][]{\citet{#2}}

% Versiones seguras para hyperref (PDF bookmarks no admiten \color)
\AtBeginDocument{%
  \pdfstringdefDisableCommands{%
    \def\DIFadd#1{#1}%
    \def\DIFdel#1{}%
    \def\DIFaddFL#1{#1}%
    \def\DIFdelFL#1{}%
    \def\DIFaddbegin{}%
    \def\DIFaddend{}%
    \def\DIFdelbegin{}%
    \def\DIFdelend{}%
  }%
  % babel-spanish activa ~ como carácter especial; dentro de \DIFadd{}/\DIFdel{}
  % seguido de } provoca "extra }". Se redefine como \nobreakspace{} estándar.
  \def~{\nobreakspace{}}%
}
