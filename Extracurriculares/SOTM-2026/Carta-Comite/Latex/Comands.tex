% Bloque del destinatario: \DatosDestinatario{nombre}{cargo}{organización}
\newcommand{\DatosDestinatario}[3]{%
  \noindent\textbf{#1}\\
  #2\\
  \textit{#3}%
}

% Línea de asunto: \Asunto{texto}
\newcommand{\Asunto}[1]{%
  \vspace{0.4cm}
  \noindent\textbf{ASUNTO:}\ #1
  \vspace{0.4cm}%
}

% Saludo de apertura: \Saludo{Estimada Mtra. Pérez:}
\newcommand{\Saludo}[1]{%
  \vspace{0.6cm}
  \noindent #1
  \vspace{0.3cm}%
}

% Bloque de firma simple: \FirmaBloque{nombre}{cargo}
\newcommand{\FirmaBloque}[2]{%
  \vspace{1.2cm}
  \noindent\textbf{Atentamente,}\\[2.0cm]
  \noindent\rule{0.45\textwidth}{0.6pt}\\[0.2cm]
  \noindent\textbf{#1}\\
  \noindent{\small #2}%
}

% Bloque de firma con visto bueno del asesor: \FirmaBloqueDoble{nombre1}{cargo1}{nombre2}{cargo2}
\newcommand{\FirmaBloqueDoble}[4]{%
  \vspace{1.2cm}
  \noindent\textbf{Atentamente,}\\[2.0cm]
  \begin{minipage}[t]{0.45\textwidth}
    \rule{\textwidth}{0.6pt}\\[0.2cm]
    \textbf{#1}\\
    {\small #2}
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{0.45\textwidth}
    \rule{\textwidth}{0.6pt}\\[0.2cm]
    \textbf{#2\ —\ Visto Bueno}\\% se sobreescribe con el arg real
    \textbf{#3}\\
    {\small #4}
  \end{minipage}%
}

% Separador de sección dentro de carta: \SeccionCarta{texto}
\newcommand{\SeccionCarta}[1]{%
  \vspace{0.5cm}
  {\small\bfseries\MakeUppercase{#1}}\\[-0.1cm]
  \rule{0.3\textwidth}{0.4pt}
  \vspace{0.2cm}%
}

% Referencia cruzada estilizada: \RefDoc{MU-FESA-2026-01}
\newcommand{\RefDoc}[1]{%
  {\small\texttt{#1}}%
}
