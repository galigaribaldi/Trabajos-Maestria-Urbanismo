% Bloque del destinatario: \DatosDestinatario{nombre}{cargo}{organización}
\newcommand{\DatosDestinatario}[3]{%
  \noindent\textbf{#1}\\
  #2\\
  \textit{#3}%
}

% Línea de asunto: \Asunto{texto}
\newcommand{\Asunto}[1]{%
  \vspace{0.4cm}
  \noindent\textbf{ASUNTO:}\ #1
  \vspace{0.4cm}%
}

% Saludo de apertura: \Saludo{Estimada Mtra. Pérez:}
\newcommand{\Saludo}[1]{%
  \vspace{0.6cm}
  \noindent #1
  \vspace{0.3cm}%
}

% Bloque de firma: \FirmaBloque{nombre}{cargo}
\newcommand{\FirmaBloque}[2]{%
  \vspace{1.2cm}
  \noindent\textbf{Atentamente,}\\[2.0cm]
  \noindent{\color{ColorPrincipal}\rule{0.45\textwidth}{0.6pt}}\\[0.2cm]
  \noindent\textbf{#1}\\
  \noindent{\small #2}\\
  \noindent{\small\color{ColorAcento}Apimetro}%
}

% Separador de sección dentro de carta: \SeccionCarta{texto}
\newcommand{\SeccionCarta}[1]{%
  \vspace{0.5cm}
  {\small\bfseries\color{ColorPrincipal}\MakeUppercase{#1}}\\[-0.1cm]
  {\color{ColorPrincipal!40}\rule{0.3\textwidth}{0.4pt}}
  \vspace{0.2cm}%
}

% Referencia cruzada estilizada: \RefDoc{APM-2026-01}
\newcommand{\RefDoc}[1]{%
  {\small\color{ColorPrincipal}\texttt{#1}}%
}
